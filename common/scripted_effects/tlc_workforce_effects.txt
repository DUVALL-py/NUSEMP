# Trekker's Victoria: Long Century
# Workforce & Research Scripted Effects

# ═══════════════════════════════════════════════════════════════
# WORKFORCE ERA APPLICATION
# ═══════════════════════════════════════════════════════════════

# Apply the correct workforce era modifier based on current date
tlc_apply_workforce_era = {
	# Remove all existing era modifiers first
	remove_modifier = tlc_workforce_era_2
	remove_modifier = tlc_workforce_era_3
	remove_modifier = tlc_workforce_era_4
	
	# Era 4: 1925-1945
	if = {
		limit = { game_date >= 1925.1.1 }
		add_modifier = {
			name = tlc_workforce_era_4
			months = -1
		}
	}
	# Era 3: 1900-1925
	else_if = {
		limit = { game_date >= 1900.1.1 }
		add_modifier = {
			name = tlc_workforce_era_3
			months = -1
		}
	}
	# Era 2: 1870-1900
	else_if = {
		limit = { game_date >= 1870.1.1 }
		add_modifier = {
			name = tlc_workforce_era_2
			months = -1
		}
	}
	# Era 1: 1836-1870 — no modifier needed (baseline)
}
