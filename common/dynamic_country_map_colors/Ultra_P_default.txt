prussian_germany = {
	color = "prussian_blue_color"

	possible = {
		OR = {
			exists = c:GER
			exists = c:NGF
		}
		OR = {
			AND = {
				exists = c:GER
				THIS = c:GER
			}
			AND = {
				exists = c:NGF
				THIS = c:NGF
			}
			AND = {
				exists = c:PRU
				THIS = c:PRU
			}
		}
		THIS = {
			OR = {
				any_interest_group = {
					has_ideology = ideology:ideology_junker_stratocratic
				}
				any_interest_group = {
					has_ideology = ideology:ideology_junker_paternalistic 
				}
			}
		}
	}
}
