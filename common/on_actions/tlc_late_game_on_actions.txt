# Trekker's Victoria: Long Century
# Late Game On Actions — Reduce processing burden after 1900

on_year_pulse = {
	effect = {
		# After 1900: More aggressive dormancy for small AI
		if = {
			limit = { game_date >= 1900.1.1 }
			every_country = {
				limit = {
					is_player = no
					NOT = { country_rank = rank_value:great_power }
					NOT = { country_rank = rank_value:major_power }
					is_at_war = no
					country_population < 1000000
				}
				# Force smaller countries into dormancy more aggressively
				if = {
					limit = { NOT = { has_modifier = tlc_ai_dormant } }
					tlc_set_dormant = yes
				}
			}
		}
		
		# After 1925: Even more aggressive
		if = {
			limit = { game_date >= 1925.1.1 }
			every_country = {
				limit = {
					is_player = no
					NOT = { country_rank = rank_value:great_power }
					is_at_war = no
					country_population < 2000000
					number_of_states < 5
				}
				if = {
					limit = {
						NOT = { has_modifier = tlc_ai_dormant }
						NOT = { has_modifier = tlc_ai_limited }
					}
					tlc_set_limited = yes
				}
			}
		}
	}
}
