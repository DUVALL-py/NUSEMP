# ——— LAISSEZ-FAIRE CAPITALISM ———
gov_tlc_laissez_faire = {
	transfer_of_power = election
	
	male_ruler = "President"
	female_ruler = "President"
	
	possible = {
		has_law = law_type:law_presidential_republic
		OR = {
			has_law = law_type:law_laissez_faire
			has_law = law_type:law_free_trade
		}
		NOT = { has_law = law_type:law_command_economy }
	}
	
	on_government_type_change = {
		effect = {
			add_modifier = {
				name = tlc_laissez_faire_bonus
				months = -1
			}
		}
	}
}

# ——— ULTRA-CAPITALIST STATE ———
gov_tlc_ultra_capitalism = {
	transfer_of_power = election
	
	male_ruler = "Chairman"
	female_ruler = "Chairwoman"
	
	possible = {
		OR = {
			has_law = law_type:law_presidential_republic
			has_law = law_type:law_parliamentary_republic
		}
		has_law = law_type:law_laissez_faire
		has_law = law_type:law_free_trade
		NOT = { has_law = law_type:law_command_economy }
	}
	
	on_government_type_change = {
		effect = {
			add_modifier = {
				name = tlc_ultra_capitalism_benefits
				months = -1
			}
			add_modifier = {
				name = tlc_ultra_capitalism_costs
				months = -1
			}
		}
	}
}
