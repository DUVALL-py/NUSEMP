# Player-facing diplomatic action. Its availability and maintenance rules mirror vanilla embargo.
cje_joint_embargo_action = {
	groups = { general }
	requires_approval = no
	should_notify_third_parties = yes

	unlocking_technologies = {
		international_trade
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_revolutionary = no
		}
	}

	possible = {
		NOT = {
			scope:target_country = {
				has_treaty_trade_privilege_with = { TARGET = root }
			}
		}
		NOT = {
			scope:target_country = {
				has_treaty_foreign_investment_rights_with = { TARGET = root }
			}
		}
		NOT = {
			custom_tooltip = {
				text = "power_bloc_disallow_embargo_bool"
				is_in_same_power_bloc = scope:target_country
				power_bloc ?= {
					modifier:power_bloc_disallow_embargo_bool = yes
				}
			}
		}
		custom_tooltip = {
			text = cje_embargo_mutual_exclusion_tt
			NOT = {
				has_diplomatic_pact = {
					who = scope:target_country
					type = embargo
				}
			}
			NOT = {
				has_diplomatic_pact = {
					who = scope:target_country
					type = cje_joint_embargo
				}
			}
		}
	}

	accept_effect = {
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = root
			SECOND = scope:target_country
			PRO_AMOUNT = -2
			ANTI_AMOUNT = 2
			FACTOR = appeasement_embargo_declared
		}
	}

	pact = {
		cost = 150
		forced_duration = 12

		is_hostile = yes
		is_embargo = yes
		is_two_sided_pact = no

		max_target_involvement = 1000
		target_involvement_applies_to = mutual
		relations_progress_per_day = -1

		first_modifier = {
			country_pact_leverage_generation_mult = -0.5
		}

		requirement_to_maintain = {
			trigger = {
				has_diplomatic_relevance = scope:target_country
			}
		}
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_NOT_DECENTRALIZED"
					NOT = { is_country_type = decentralized }
					scope:target_country = {
						NOT = { is_country_type = decentralized }
					}
				}
			}
		}
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_DIPLOMATIC_AUTONOMY"
					aggressive_diplomatic_plays_permitted = yes
					scope:target_country = { aggressive_diplomatic_plays_permitted = yes }
				}
			}
		}
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_NO_SUBJECT_RELATION"
					NOT = {
						is_subject_of = scope:target_country
						scope:target_country = { is_subject_of = root }
					}
				}
			}
		}
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = DIPLO_ACTION_RELATIONS_POOR_OR_LESS_NOT
					scope:target_country.relations:root <= relations_threshold:poor
				}
			}
			show_about_to_break_warning = {
				scope:target_country.relations:root >= -23
				scope:target_country.relations_change_rate:root > 0
			}
		}
		requirement_to_maintain = {
			trigger = { country_has_isolationism_or_sakoku = no }
		}
		requirement_to_maintain = {
			trigger = {
				scope:target_country = { country_has_isolationism_or_sakoku = no }
			}
		}
		requirement_to_maintain = {
			trigger = { is_junior_in_customs_union = no }
		}
		requirement_to_maintain = {
			trigger = { scope:target_country = { is_junior_in_customs_union = no } }
		}
		requirement_to_maintain = {
			trigger = {
				NOT = { scope:target_country = { has_treaty_port_in_market = root.market } }
			}
		}
		requirement_to_maintain = {
			trigger = {
				NOT = {
					scope:target_country = {
						has_treaty_foreign_investment_rights_with = { TARGET = root }
					}
				}
			}
		}

		manual_break_effect = {
			add_lobby_appeasement_from_diplomacy_unidirectional = {
				FIRST = root
				SECOND = scope:target_country
				PRO_AMOUNT = 2
				ANTI_AMOUNT = -2
				FACTOR = appeasement_embargo_ended
			}
		}
		auto_break_effect = {
			add_lobby_appeasement_from_diplomacy_unidirectional = {
				FIRST = root
				SECOND = scope:target_country
				PRO_AMOUNT = 2
				ANTI_AMOUNT = -2
				FACTOR = appeasement_embargo_ended
			}
		}
	}

	ai = {
		check_acceptance_for_will_break = no
		max_influence_spending_fraction = 0.2
		evaluation_chance = {
			value = 0
			if = {
				limit = { total_import_value >= 15000 }
				add = 0.01
				if = {
					limit = { total_import_value >= 100000 }
					add = 0.05
				}
				else_if = {
					limit = { total_import_value >= 30000 }
					add = 0.025
				}
			}
		}
		propose_score = {
			value = 10
			if = {
				limit = {
					AND = {
						has_radical_leftist_government = no
						scope:target_country = { has_radical_leftist_government = yes }
					}
				}
				multiply = 2
			}
			if = {
				limit = {
					AND = {
						has_radical_leftist_economy = no
						scope:target_country = { has_radical_leftist_economy = yes }
					}
				}
				multiply = 2
			}
		}
		will_propose = {
			"scope:target_country.mutual_trade_value_with_country(root)" > 0
			NOT = {
				has_attitude = { who = scope:target_country attitude = conciliatory }
			}
			OR = {
				scope:target_country.relations:root <= relations_threshold:hostile
				scope:target_country.infamy >= infamy_threshold:pariah
				scope:target_country = {
					has_diplomatic_pact = { who = root type = cje_joint_embargo_action }
				}
			}
		}
		will_break = {
			OR = {
				has_attitude = { who = scope:target_country attitude = conciliatory }
				NOR = {
					scope:target_country.relations:root <= relations_threshold:hostile
					scope:target_country.infamy >= infamy_threshold:pariah
					scope:target_country = {
						has_diplomatic_pact = { who = root type = cje_joint_embargo_action }
					}
				}
			}
		}
	}
}

# Hidden follower relationship created by a Joint Embargo diplomatic action.
cje_joint_embargo = {
	groups = { general }
	requires_approval = no
	should_notify_third_parties = no
	selectable = { always = no }
	potential = { NOT = { this = scope:target_country } }

	pact = {
		cost = 0
		is_hostile = yes
		is_embargo = yes
		is_two_sided_pact = no
		max_target_involvement = 0
		target_involvement_applies_to = mutual

		actor_can_break = { always = no }
		target_can_break = { always = no }

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = cje_joint_embargo_must_follow_tt
					hidden_trigger = {
						any_country = {
							is_country_alive = yes
							has_diplomatic_pact = {
								who = scope:target_country
								type = cje_joint_embargo_action
							}
							OR = {
								has_treaty_alliance_with = { TARGET = ROOT }
								any_subject_or_below = { this = ROOT }
							}
						}
					}
				}
			}
		}

		manual_break_effect = { }
		auto_break_effect = { }
	}

	ai = {
		evaluation_chance = { value = 0 }
		will_propose = { always = no }
		will_break = { always = no }
	}
}
