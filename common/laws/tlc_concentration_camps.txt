# Trekker's Victoria: Long Century
# Concentration Camps — Political/historical mechanic
# Requires: ethnostate + one-party state
# Implemented via laws + modifiers — NO Pop loops

# ═══════════════════════════════════════════════════════════════
# LAW GROUP
# ═══════════════════════════════════════════════════════════════

law_group_tlc_internment_policy = {
	law_group_category = rights
	
	progressiveness = 0
	
	default = law_tlc_no_internment
}

# ═══════════════════════════════════════════════════════════════
# NO INTERNMENT — Default state (no camps)
# ═══════════════════════════════════════════════════════════════

law_tlc_no_internment = {
	group = law_group_tlc_internment_policy
	
	icon = "gfx/interface/icons/law_icons/no_police.dds"
	
	progressiveness = 0
	
	# No modifiers — this is the default civilized state
	
	possible_political_movements = {
		law_tlc_internment_camps
	}
	
	ai_will_do = {
		value = 100
	}
}

# ═══════════════════════════════════════════════════════════════
# INTERNMENT CAMPS — Requires specific political conditions
# ═══════════════════════════════════════════════════════════════

law_tlc_internment_camps = {
	group = law_group_tlc_internment_policy
	
	icon = "gfx/interface/icons/law_icons/secret_police.dds"
	
	progressiveness = -100
	
	# STRICT REQUIREMENTS — ethnostate + authoritarian
	unlocking_laws = {
		law_ethnostate
		law_outlawed_dissent
	}
	
	modifier = {
		# "Benefits" for the regime
		country_authority_mult = 0.10
		state_radicals_from_sol_change_mult = -0.15
		
		# Severe costs
		country_prestige_mult = -0.20
		state_migration_pull_mult = -0.30
		country_tech_research_speed_mult = -0.05
		state_birth_rate_mult = -0.05
	}
	
	possible_political_movements = {
		law_tlc_no_internment
	}
	
	ai_will_do = {
		value = 0
		# Only extreme AI governments will consider this
		if = {
			limit = {
				has_law = law_type:law_ethnostate
				has_law = law_type:law_outlawed_dissent
			}
			add = 25
		}
	}
}
