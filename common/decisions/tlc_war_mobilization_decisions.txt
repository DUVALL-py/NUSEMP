# Trekker's Victoria: Long Century — War Mobilization & Defense Decisions

tlc_declare_general_mobilization = {
	is_shown = {
		is_at_war = yes
		NOT = { has_modifier = tlc_general_mobilization_modifier }
	}

	possible = {
		is_at_war = yes
	}

	when_taken = {
		add_modifier = {
			name = tlc_general_mobilization_modifier
			months = 24
		}
		add_war_exhaustion = 5
	}

	ai_chance = {
		base = 50
		modifier = {
			trigger = {
				any_enemy_country = {
					country_rank >= rank_value:major_power
				}
			}
			add = 50
		}
	}
}
