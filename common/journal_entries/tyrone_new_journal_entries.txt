je_contain_epidemic = {
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	group = je_group_internal_affairs

	is_shown_when_inactive = yes

	possible = {
		is_player = yes
		has_variable = active_epidemic
	}

	complete = {
		NOT = { has_variable = active_epidemic }
	}

	on_complete = {
		add_modifier = {
			name = modifier_epidemic_conquered
			months = 120
		}
	}
}

je_railway_era = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_industrialization

	is_shown_when_inactive = yes

	possible = {
		is_player = yes
		has_technology = railways
	}

	complete = {
		num_buildings = {
			type = building_railway
			value >= 20
		}
	}

	on_complete = {
		add_modifier = {
			name = modifier_railway_boom
			months = 240
		}
		add_prestige = 100
	}
}
