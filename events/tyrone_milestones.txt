namespace = tyrone_milestone

# First Railway Built
tyrone_milestone.1 = {
	type = country_event
	placement = ROOT
	title = tyrone_milestone.1.t
	desc = tyrone_milestone.1.d
	flavor = tyrone_milestone.1.f
	event_image = { video = "unspecific_science" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		is_player = yes
		has_technology_researched = railways
		any_scope_building = { is_building_type = building_railway }
		NOT = { has_variable = milestone_first_railway }
	}

	immediate = {
		set_variable = milestone_first_railway
	}

	option = {
		name = tyrone_milestone.1.a
		default_option = yes
		add_modifier = { name = modifier_railway_pioneer }
	}
}
