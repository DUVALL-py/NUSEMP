# Trekker's Victoria: Long Century
# Workforce Events — Era transition notifications

namespace = tlc_workforce

# ═══════════════════════════════════════════════════════════════
# ERA TRANSITION EVENTS — Triggered by date, applied globally
# ═══════════════════════════════════════════════════════════════

# Era 2: 1870 — Industrialization begins
tlc_workforce.1 = {
	type = country_event
	placement = root
	
	title = tlc_workforce.1.t
	desc = tlc_workforce.1.d
	flavor = tlc_workforce.1.f
	
	event_image = {
		texture = "gfx/event_pictures/politics_social_reform.dds"
	}
	
	trigger = {
		game_date >= 1870.1.1
		game_date < 1871.1.1
		is_player = yes
		NOT = { has_variable = tlc_workforce_era_2_notified }
	}
	
	immediate = {
		set_variable = tlc_workforce_era_2_notified
		tlc_apply_workforce_era = yes
	}
	
	option = {
		name = tlc_workforce.1.a
		default_option = yes
	}
}
