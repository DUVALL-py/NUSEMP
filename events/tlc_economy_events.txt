# Era 2: 1860 — Research begins slowing
tlc_economy.1 = {
	type = country_event
	placement = root
	
	title = tlc_economy.1.t
	desc = tlc_economy.1.d
	flavor = tlc_economy.1.f
	
	event_image = {
		texture = "gfx/event_pictures/politics_academia.dds"
	}
	
	trigger = {
		game_date >= 1860.1.1
		game_date < 1861.1.1
		is_player = yes
		NOT = { has_variable = tlc_research_era_2_notified }
	}
	
	immediate = {
		set_variable = tlc_research_era_2_notified
		tlc_apply_research_era = yes
	}
	
	option = {
		name = tlc_economy.1.a
		default_option = yes
	}
}
