namespace = tyrone_flavor

tyrone_flavor.1 = {
	type = country_event
	placement = ROOT

	title = tyrone_flavor.1.t
	desc = tyrone_flavor.1.d
	flavor = tyrone_flavor.1.f

	event_image = {
		video = "unspecific_world_fair"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		has_technology_researched = camera
		is_player = yes
		NOT = { has_variable = had_first_photograph }
	}

	immediate = {
		set_variable = had_first_photograph
	}

	option = {
		name = tyrone_flavor.1.a
		default_option = yes
		add_modifier = { name = modifier_photographic_innovation months = 120 }
	}

	option = {
		name = tyrone_flavor.1.b
		add_treasury = 25000
	}
}
