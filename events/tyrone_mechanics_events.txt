namespace = tyrone_mechanics

tyrone_mechanics.1 = {
	type = country_event
	placement = ROOT

	title = tyrone_mechanics.1.t
	desc = tyrone_mechanics.1.d
	flavor = tyrone_mechanics.1.f

	event_image = {
		video = "unspecific_devastation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		is_player = yes
		game_date >= 1840.1.1
		NOT = { has_variable = had_cholera }
	}

	immediate = {
		set_variable = had_cholera
		set_variable = active_epidemic
	}

	option = {
		name = tyrone_mechanics.1.a
		default_option = yes
		add_modifier = {
			name = modifier_local_quarantine
			months = 36
		}
	}

	option = {
		name = tyrone_mechanics.1.b
		add_radicals = {
			value = 0.05
			strata = poor
		}
	}
}
