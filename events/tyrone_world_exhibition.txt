namespace = tyrone_exhibition

tyrone_exhibition.1 = {
	type = country_event
	placement = ROOT

	title = tyrone_exhibition.1.t
	desc = tyrone_exhibition.1.d
	flavor = tyrone_exhibition.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	option = {
		name = tyrone_exhibition.1.a
		default_option = yes
		add_treasury = -500000
		set_variable = { name = exhibition_quality value = 3 }
		trigger_event = { id = tyrone_exhibition.2 months = 6 }
	}

	option = {
		name = tyrone_exhibition.1.b
		add_treasury = -200000
		set_variable = { name = exhibition_quality value = 1 }
		trigger_event = { id = tyrone_exhibition.2 months = 6 }
	}
}

tyrone_exhibition.2 = {
	type = country_event
	placement = ROOT

	title = tyrone_exhibition.2.t
	desc = tyrone_exhibition.2.d
	flavor = tyrone_exhibition.2.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	immediate = {
		if = {
			limit = {
				has_variable = exhibition_quality
				var:exhibition_quality >= 3
			}
			add_modifier = { name = modifier_world_exhibition_grand months = 120 }
		}
		else = {
			add_modifier = { name = modifier_world_exhibition_modest months = 60 }
		}
	}

	option = {
		name = tyrone_exhibition.2.a
		default_option = yes
	}
}
