namespace = trekking_news

# Campaign opening newspaper: "O Jogo Vai Começar / The Game is Starting"
trekking_news.1 = {
	type = country_event
	placement = ROOT

	title = trekking_news.1.t
	desc = trekking_news.1.d
	flavor = trekking_news.1.f

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		is_player = yes
		game_date < 1837.1.1
		NOT = { has_variable = trekking_news_campaign_opened }
	}

	immediate = {
		set_variable = trekking_news_campaign_opened
	}

	option = {
		name = trekking_news.1.a
		default_option = yes
	}
}

# German unification world headline
trekking_news.10 = {
	type = country_event
	placement = ROOT

	title = trekking_news.10.t
	desc = trekking_news.10.d
	flavor = trekking_news.10.f

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		is_player = yes
		exists = c:GER
		NOT = { THIS = c:GER }
		NOT = { has_variable = trekking_news_germany_proclaimed }
	}

	immediate = {
		set_variable = trekking_news_germany_proclaimed
	}

	option = {
		name = trekking_news.10.a
		default_option = yes
		c:GER ?= {
			change_relations = {
				country = ROOT
				value = 15
			}
		}
		add_modifier = {
			name = modifier_formation_warm_diplomatic_ties
			months = 60
		}
	}

	option = {
		name = trekking_news.10.b
		c:GER ?= {
			change_relations = {
				country = ROOT
				value = -20
			}
		}
		add_modifier = {
			name = modifier_formation_cold_diplomatic_ties
			months = 60
		}
	}
}
